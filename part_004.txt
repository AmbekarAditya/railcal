    const monthNumber = monthNames[selectedMonth];
    const firstTripDate = duty.trips.length > 0 ? duty.trips[0].date : '';
    const defaultStartDate = firstTripDate ? new Date(firstTripDate) : new Date(currentYear, monthNumber - 1, 1);

    let defaultEndDate = new Date(defaultStartDate);
    let defaultEndTime = new Date(defaultStartDate);
  
    if (duty.trips.length > 0) {

      if (lastTripDepartureTime) {

        defaultEndTime.setHours(parseInt(hours, 10));
        defaultEndTime.setMinutes(parseInt(minutes, 10) + 30);
  
        // Adjust the defaultEndDate if the defaultEndTime crosses midnight
        if (defaultEndTime < defaultStartDate) {
          defaultEndDate.setDate(defaultEndDate.getDate() + 1);

const ArrivalDepartureTimePicker: React.FC<ArrivalDepartureTimePickerProps> = ({
  onTimesChange,
  disabled,
}) => {
  const [arrivalTime, setArrivalTime] = useState("");
  const [departureTime, setDepartureTime] = useState("");
  const [showArrivalPicker, setShowArrivalPicker] = useState(false);
  const [showDeparturePicker, setShowDeparturePicker] = useState(false);

  useEffect(() => {
    if (arrivalTime && departureTime) {
      const timeDiff = calculateTimeDifference(arrivalTime, departureTime);
      onTimesChange(arrivalTime, departureTime, timeDiff);
    }
  }, [arrivalTime, departureTime]);

  const calculateTimeDifference = (arrival: string, departure: string) => {
    const arrivalDate = new Date(`2021/01/01 ${arrival}`);
    const departureDate = new Date(`2021/01/01 ${departure}`);

    // Check if arrival time is less than departure time
    if (arrivalDate.getTime() > departureDate.getTime()) {
      // Increment the departure date by one day
      departureDate.setDate(departureDate.getDate() + 1);
    }

    const timeDiff = getTimeDifference(arrivalDate, departureDate);
    const hours = Math.floor(timeDiff / 60);
    const minutes = timeDiff % 60;
    return `${hours}h ${minutes}m`;
  };

  const getTimeDifference = (
    arrivalTime: Date,
    departureTime: Date
  ): number => {
    if (!arrivalTime || !departureTime) {
      return 0;
    }

    // Check if arrival and departure times are equal
    if (arrivalTime.getTime() === departureTime.getTime()) {
      // Special case: both times are at midnight
      if (arrivalTime.getHours() === 0 && arrivalTime.getMinutes() === 0) {
        return 24 * 60;
      }
      return 0;
    }

    if (
      arrivalTime.getHours() === 0 &&
      arrivalTime.getMinutes() === 0 &&
      departureTime.getHours() === 0 &&
      departureTime.getMinutes() === 0
    ) {
      return 24 * 60;
    }

    const timeDiff = Math.round(
      (departureTime.getTime() - arrivalTime.getTime()) / 60000
    );

    // Return 0 if the time difference is negative
    return timeDiff >= 0 ? timeDiff : 0;
  };

  const handleArrivalTimeChange = (
    event: React.ChangeEvent<HTMLInputElement>
  ) => {
    setArrivalTime(event.target.value);
  };

  const handleDepartureTimeChange = (
    event: React.ChangeEvent<HTMLInputElement>
  ) => {
    setDepartureTime(event.target.value);
  };

  const openArrivalPicker = () => {
    setShowArrivalPicker(true);
  };

  const openDeparturePicker = () => {
    setShowDeparturePicker(true);
  };

  const handleTimeChange =
    (isArrival: boolean) => (event: any, selectedTime?: Date) => {
      if (event.type === "set" && selectedTime) {
        const hour = String(selectedTime.getHours()).padStart(2, "0");
        const minute = String(selectedTime.getMinutes()).padStart(2, "0");
        const timeString = `${hour}:${minute}`;

        if (isArrival) {
          setArrivalTime(timeString);
          setShowArrivalPicker(false);
        } else {
          setDepartureTime(timeString);
          setShowDeparturePicker(false);
        }
      } else {
        setShowArrivalPicker(false);
        setShowDeparturePicker(false);
      }
    };

  const renderTimePicker = (isArrival: boolean) => {
    const time = isArrival ? arrivalTime : departureTime;
    const setShowPicker = isArrival
      ? setShowArrivalPicker
      : setShowDeparturePicker;
    const showPicker = isArrival ? showArrivalPicker : showDeparturePicker;

      <View style={styles.timePickerContainer}>

            disabled={disabled}
          >
            <View style={styles.timeContainer}>
              <Text style={styles.timeText}>{time || "--:--"}</Text>
            </View>
          </TouchableOpacity>
        )}

            value={new Date()}
            mode="time"
            is24Hour={true}
            display="default"
            onChange={handleTimeChange(isArrival)}
          />
        )}
      </View>
    );
  };

  return (
    <View style={styles.container}>
      {renderTimePicker(true)}
      {renderTimePicker(false)}
    </View>

class Duty {
  id: number;
  trips: Trip[];
  signIn: Date | null;
  signOff: Date | null;
  dutyHours: number;
  nightDutyHours: number;
  isOSDA: boolean;
  isRRA: boolean;
  dutyAllowance: number;
  dutyDistance: number;
  progressiveDutyHours:number;
  progressiveNightDutyHours: number;
  hasEnded: boolean; // Add this line
  isLeave: boolean;

  constructor(id: number, trips: Trip[] = []) {
    this.id = id;
    this.trips = trips;
    this.signIn = null;
    this.signOff = null;
    this.dutyHours = 0;
    this.nightDutyHours = 0;
    this.isOSDA = false;
    this.isRRA = false;
    this.dutyAllowance = 0;
    this.dutyDistance = 0;
    this.progressiveDutyHours=0;
    this.progressiveNightDutyHours=0;
    this.hasEnded = false; // Initialize hasEnded
    this.isLeave = false;
  }

  setSignInAndSignOff(signIn: Date, signOff: Date) {
    this.signIn = signIn;
    this.signOff = signOff;
    this.calculateDutyHoursAndNightDutyHours();
  }

  private calculateDutyHoursAndNightDutyHours() {
    if (this.signIn && this.signOff) {
      this.dutyHours = this.calculateDutyHours();
      this.nightDutyHours = this.calculateNightDutyHours();
    // this.progressiveNightDutyHours += this.nightDutyHours;
    }
  }

  calculateTotalAllowance() {
    let totalAllowanceforDuty = 0;
  
    // Calculate the total allowance for non-leave trips
    const nonLeaveTripAllowances = this.trips
      .filter(trip => !["Paid leave", "Sick leave", "Casual leave"].includes(trip.typeOfWork))
      .map(trip => trip.allowance);
  
    totalAllowanceforDuty = nonLeaveTripAllowances.reduce((total, allowance) => total + allowance, 0);
  
    // Apply the minimum allowance of 120 only for non-leave trips
    if (totalAllowanceforDuty > 0 && totalAllowanceforDuty <= 120) {
      totalAllowanceforDuty = 120;
    }
  
    this.dutyAllowance = totalAllowanceforDuty;
    return totalAllowanceforDuty;
  }
  calculateTotalDistance() {
    const totalDistance = this.trips.reduce((acc, trip) => acc + (trip.distance || 0), 0);
    return totalDistance;
  }

  // calculateprogressiveDutyHours(dutyIndex: number, totalDuties: Duty[]): number {

  calculateProgressiveNightDutyHours(dutyIndex: number, totalDuties: Duty[]): number {
    if (dutyIndex === 0) {
      // For the first duty, the progressiveNightDutyHours is the same as nightDutyHours
      this.progressiveNightDutyHours = this.nightDutyHours;
    } else {
      // Calculate the progressive night duty hours based on the previous duty's progressiveNightDutyHours
      const previousDuty = totalDuties[dutyIndex - 1];
      this.progressiveNightDutyHours = previousDuty.progressiveNightDutyHours + this.nightDutyHours;
    }

    return this.progressiveNightDutyHours;
}

updateDutyHoursForLeaveTypes(typeOfWork: string) {
  const endedTypes = ["Periodic rest", "Waiting for duty"];

  if (leaveTypes.includes(typeOfWork)) {
    this.dutyHours = 7.41667; // 7 hours and 25 minutes
    this.dutyAllowance = 0;
    this.dutyDistance = 0;
    this.isLeave = true;
  } else if (endedTypes.includes(typeOfWork)) {
    this.dutyHours = 0; // Set dutyHours to 0 for "Periodic Rest" and "Waiting for duty"
    this.dutyAllowance = 0;
    this.dutyDistance = 0;
  } else {
    this.dutyHours = this.calculateDutyHours();
    this.dutyAllowance = this.calculateTotalAllowance();
    this.dutyDistance = this.calculateTotalDistance();
  }
}
  updateTrip(tripId: number, updatedTrip: Trip) {
    const tripIndex = this.trips.findIndex((trip) => trip.id === tripId);
    if (tripIndex !== -1) {
      this.trips[tripIndex] = updatedTrip;
    }
  }

  private calculateDutyHours(): number {
    if (this.signIn && this.signOff) {
      const timeDiff = Math.abs(this.signOff.getTime() - this.signIn.getTime());
      const totalMinutes = Math.floor(timeDiff / (1000 * 60));
  
      if (totalMinutes < 60) {
        return totalMinutes / 60;
      } else {
        const hours = Math.floor(totalMinutes / 60);
        const minutes = totalMinutes % 60;
        const fractionalHours = minutes / 60;
        return hours + fractionalHours;
      }
    }
    return 0;
  }
  
  private calculateNightDutyHours(): number {
    if (this.signIn && this.signOff) {
      const NIGHT_START_HOUR = 22; // 10 PM
      const NIGHT_END_HOUR = 6; // 6 AM
      let nightDutyMinutes = 0;

      // Start counting from signIn and stop at signOff
      let current = new Date(this.signIn);

      while (current < this.signOff) {
        let hour = current.getHours();
        if (hour >= NIGHT_START_HOUR || hour < NIGHT_END_HOUR) {
          nightDutyMinutes++; // Count each minute during the night hours
        }
        current.setMinutes(current.getMinutes() + 1); // Move to next minute
      }

      // Convert total minutes to hours in decimal
      let nightDutyHours = nightDutyMinutes / 60;

      return nightDutyHours;
    }

    return 0;
  }
}
