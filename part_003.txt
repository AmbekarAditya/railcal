  default: { factor: 1, remark: "" },

// TypeScript function to calculate allowance and remarks
function calculateAllowanceAndRemarks(
  distance: number,
  typeOfWork: string,
  timeInMinutes: number
): { allowance: number; remarks: string } {
  const params = caseParameters[typeOfWork] || caseParameters["default"];

  if (Array.isArray(params)) {
    for (const param of params) {
      if (distance <= param.max!) {
        return {
          allowance: parseFloat((distance * param.factor).toFixed(2)),
          remarks: param.remark,
        };
      }
    }
  } else {
    let allowance: number;
    if (params.isTimeBased) {
      // Calculate distance based on time for time-based cases
      const timeBasedConstant = getTimeBasedConstant(typeOfWork, timeInMinutes);
      const distance = (timeInMinutes / 60) * timeBasedConstant;
      allowance = parseFloat(distance.toFixed(2));
    } else {
      allowance = params.factor
        ? parseFloat((distance * params.factor).toFixed(2))
        : 0;
    }

    return {
      allowance: parseFloat(allowance.toFixed(2)),
      remarks: params.remark,
    };
  }
  return { allowance: 0, remarks: "Invalid typeOfWork or distance" }; // Fallback in case no conditions are met
}

// Helper function to get the time-based constant for the given typeOfWork
function getTimeBasedConstant(
  typeOfWork: string,
  timeInMinutes: number
): number {
  const roundedTimeInMinutes = Math.ceil(timeInMinutes / 5) * 5; // Round up to the nearest multiple of 5
  switch (typeOfWork) {
    case "Shunting":
      return 15;
    case "Light Engine on Mechanical Account":
      return 20;
    case "Breakdown Specials":
      return 25;
    case "Crane Specials":
      return 20;
    case "Material Train":
      return 20; // Special case for Material Train
    case "Medical Relief Trains":
      return 25;
    default:
      return roundedTimeInMinutes; // For other cases, return the rounded time in minutes
  }
}

function calculateLeaveMileage({
  rateOfPay,
  daysInMonth,
  duties,
}: CalculateLeaveMileageParams): CalculateLeaveMileageResult {
  // Handle missing parameters
  if (rateOfPay === undefined || daysInMonth === undefined || duties === undefined) {
    throw new Error("Missing parameters");
  }

  // Handle negative values
  if (rateOfPay < 0 || daysInMonth < 0) {
    throw new Error("Negative values are not allowed");
  }

  // Handle edge case when daysInMonth is 0
  if (daysInMonth === 0) {
    throw new Error("daysInMonth cannot be zero");
  }

  // Calculate total leaves
  const totalLeaves = duties.reduce((count, duty) => {
    return count + duty.trips.filter(trip => leaveTypes.includes(trip.typeOfWork)).length;
  }, 0);


  const lmc = (0.3 * rateOfPay) / daysInMonth;
  const leaveMileageAllowance = lmc * totalLeaves;

  return { leaveMileageAllowance };
}

import Duty from '../duty/Duty';
import Trip from '../trip/Trip';

interface Range {
  limit: number;
  multiplier: number;
  extra: number;
}

interface FortnightData {
  startDate: Date;
  endDate: Date;
  overtimeHours: number;
  overtimeMinutes: number;
  overtimeUnit: number;
}

interface OverTimeCalculationResult {
  firstFortnight: FortnightData | null;
  secondFortnight: FortnightData | null;
}

const ranges: Range[] = [
  { limit: 104 * 60, multiplier: 1.5, extra: 0 },
  { limit: 108 * 60, multiplier: 2, extra: 6 },
];

const calculateOvertimeUnit = (overtimeTotalTime: number): number => {
  let overtimeUnit = 0;

  if (overtimeTotalTime <= ranges[0].limit) {
    overtimeUnit = 0;
  } else if (overtimeTotalTime > ranges[0].limit && overtimeTotalTime <= ranges[1].limit) {
    overtimeUnit = (overtimeTotalTime - ranges[0].limit) * ranges[0].multiplier;
  } else if (overtimeTotalTime > ranges[1].limit) {
    overtimeUnit = (overtimeTotalTime - ranges[1].limit) * ranges[1].multiplier + ranges[1].extra;
  }

  return overtimeUnit;
};

const calculateFortnight = (duties: Duty[], offsetDays: number, startDate: Date | null = null): FortnightData | null => {
  if (!duties || duties.length === 0) {
    return null;
  }

  if (!startDate) {
    const firstTripDate = duties[0].trips.find(trip => trip.date)?.date;
    if (!firstTripDate) {
      console.error('calculateFortnight: Invalid first trip date');
      return null;
    }
    startDate = new Date(firstTripDate);
  }

  const endDate = new Date(startDate);
  endDate.setDate(endDate.getDate() + 13 + offsetDays);

  let overtimeTotalTime = 0;
  duties.forEach(duty => {
    if (!duty.trips || duty.trips.length === 0) {
      return;
    }

    const firstTripDate = new Date(duty.trips[0].date);
    if (isNaN(firstTripDate.getTime()) || firstTripDate < startDate || firstTripDate >= endDate) {
      return;
    }

    const overtimeForDuty = duty.dutyHours * 60 ;

    overtimeTotalTime += overtimeForDuty;
  });

  const fortnightData: FortnightData = {
    startDate: startDate,
    endDate: endDate,
    overtimeHours: Math.floor(overtimeTotalTime / 60),
    overtimeMinutes: overtimeTotalTime % 60,
    overtimeUnit: calculateOvertimeUnit(overtimeTotalTime)
  };

  return fortnightData;
};

const firstFortnight = (duties: Duty[], startDate: Date): FortnightData | null => {
  return calculateFortnight(duties, 0);
};

const secondFortnight = (duties: Duty[], firstFortnightEndDate: Date | null): FortnightData | null => {
  if (!firstFortnightEndDate || !(firstFortnightEndDate instanceof Date)) {
    return null;
  }
  const newStartDate = new Date(firstFortnightEndDate);
  newStartDate.setDate(firstFortnightEndDate.getDate() + 1);
  return calculateFortnight(duties, 0, newStartDate);
};

const overtimeCalculation = (duties: Duty[], startDate: Date): OverTimeCalculationResult => {
  
    const firstFortnightData = firstFortnight(duties, startDate);
  
    const firstFortnightEndDate = firstFortnightData ? new Date(firstFortnightData.endDate) : null;
  
    const secondFortnightData = secondFortnight(duties, firstFortnightEndDate);
  
    return {
      firstFortnight: firstFortnightData,
      secondFortnight: secondFortnightData
    };
  };

export { overtimeCalculation, calculateOvertimeUnit, calculateFortnight, firstFortnight, secondFortnight };

const PickerComponentStation: React.FC<PickerComponentStationProps> = ({
  onStationChange,
  dutyId,
  tripId,
  disabled,
  duty,
}) => {
  const [fromStation, setFromStation] = useState("");
  const [toStation, setToStation] = useState("");

  const currentTripIndex = duty.trips.findIndex((trip) => trip.id === tripId);
  const previousTrip =
    currentTripIndex > 0 ? duty.trips[currentTripIndex - 1] : null;
  const previousTripDestinationStation = previousTrip
    ? previousTrip.destinationStation
    : "";

  useEffect(() => {
    setFromStation(previousTripDestinationStation || "");
  }, [previousTripDestinationStation]);

  const handleFromStationChange = (selectedStation: string) => {
    setFromStation(selectedStation);
    updateToStationAndDistance(selectedStation, toStation);
  };

  const handleToStationChange = (selectedStation: string) => {
    setToStation(selectedStation);
    updateToStationAndDistance(fromStation, selectedStation);
  };

  const updateToStationAndDistance = (from: string, to: string) => {
    const combo = combinations.find(
      (c) => c.from_station === from && c.to_station === to
    );
    if (combo) {
      onStationChange(dutyId, tripId, from, to, combo.distance);
    }
  };

  const uniqueFromStations = Array.from(
    new Set(combinations.map((item) => item.from_station))
  );

  const renderPicker = (isFrom: boolean) => {
    const label = isFrom ? "From:" : "To:";
    const value = isFrom ? fromStation : toStation;
    const onValueChange = isFrom ? handleFromStationChange : handleToStationChange;
    const items = isFrom
      ? uniqueFromStations
      : combinations.filter((item) => item.from_station === fromStation).map((item) => item.to_station);

    return (
      <View style={styles.pickerContainer}>
        <Text style={styles.label}>{label}</Text>
        {Platform.OS === "web" ? (
          <select
            value={value}
            onChange={(e) => onValueChange(e.target.value)}
            disabled={disabled}
            style={styles.webSelect}
          >

                {station}
              </option>
            ))}

        ) : (
          <Picker
            selectedValue={value}
            onValueChange={onValueChange}
            enabled={!disabled}
            style={styles.picker}
          >
            <Picker.Item label={isFrom ? "From Station" : "To Station"} value="" />
            {items.map((station, index) => (
              <Picker.Item key={index} label={station} value={station} />
            ))}
          </Picker>
        )}
      </View>
    );
  };

  const [fortnights, setFortnights] = useState<Fortnight[]>([]);
  const [pastFortnightHours, setPastFortnightHours] = useState(0);
  const [pastFortnightMinutes, setPastFortnightMinutes] = useState(0);

  useEffect(() => {
    const pastFortnight = pastFortnightHours + pastFortnightMinutes / 60;
    recalculateFortnights(pastFortnight);
    onFortnightsChange(fortnights);
  }, [selectedMonth, duties, pastFortnightHours, pastFortnightMinutes]);

  const recalculateFortnights = (pastFortnight: number) => {
    const monthIndex = new Date(
      Date.parse(`${selectedMonth} 1, 2024`)
    ).getMonth();
    const startDate = new Date(2024, monthIndex, 1);
    const endDate = new Date(2024, monthIndex + 1, 0);

      // Sort duties by start date in ascending order
  const sortedDuties = [...duties].sort((a, b) => {
    const aStartDate = a.signIn ? new Date(a.signIn) : new Date(a.trips[0].date);
    const bStartDate = b.signIn ? new Date(b.signIn) : new Date(b.trips[0].date);
    return aStartDate.getTime() - bStartDate.getTime();
  });

    const intersectingFortnights: Fortnight[] = FORTNIGHT_DATES.filter(
      (date) => {
        const fortnightEnd = new Date(date);
        fortnightEnd.setDate(fortnightEnd.getDate() + 13);
        return date <= endDate && fortnightEnd >= startDate;
      }
    ).map((startDate, index) => {
      const endDate = new Date(startDate.getTime() + 13 * 24 * 60 * 60 * 1000);

      let totalForthnightDuration = index === 0 ? pastFortnight : 0;

      sortedDuties.forEach((duty) => {
        let dutyStart;
        if (duty.isLeave && duty.trips.length > 0) {
          dutyStart = new Date(duty.trips[0].date);
        } else {
          dutyStart = new Date(duty.signIn);
        }
        if (dutyStart >= startDate && dutyStart <= endDate) {
          const dutyHours =
            duty.dutyHours && !isNaN(duty.dutyHours)
              ? parseFloat(duty.dutyHours)
              : 0;
          totalForthnightDuration += dutyHours;
          // if (!duty.isLeave) {
          duty.progressiveDutyHours = totalForthnightDuration;
        }
      });

      const basicPlusDAPlusTA =
        rateOfPay +
        dearnessAllowance +
        transportationAllowance +
        rateOfPay * 0.3;
      const lowerRate = (basicPlusDAPlusTA * 1.5) / 223;
      const higherRate = (basicPlusDAPlusTA * 2) / 223;

      let overtimeAllowance = 0;
      let totalOvertimeHours = 0;
  
      if (totalForthnightDuration > 104) {
        const overtimeHours = totalForthnightDuration - 104;
  
        if (overtimeHours <= 4) {
          overtimeAllowance = overtimeHours * lowerRate;
        } else {
          overtimeAllowance = 4 * lowerRate + (overtimeHours - 4) * higherRate;
        }
  
        totalOvertimeHours = overtimeHours;
      }

      return new Fortnight(
        startDate,
        endDate,
        totalForthnightDuration,
        overtimeAllowance,
        totalOvertimeHours
      );
    });

    const totalOvertimeAllowance = intersectingFortnights.reduce(
      (total, fortnight) => total + fortnight.overtimeAllowance,
      0
    );

    onOvertimeAllowanceChange(totalOvertimeAllowance);
  };

  //   setInputHours(hours);
  // };

  const handlePastFortnightHoursChange = (text: string) => {

      setPastFortnightHours(hours);
    } else {
      setPastFortnightHours(0);
    }
  };

  const handlePastFortnightMinutesChange = (text: string) => {
    const minutes = parseInt(text, 10);
    if (!isNaN(minutes) && minutes >= 0 && minutes <= 59) {
      setPastFortnightMinutes(minutes);
    } else {
      setPastFortnightMinutes(0);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Fortnights in {selectedMonth}</Text>
      <View style={styles.inputContainer}>
        <Text style={styles.inputLabel}>Past Fortnight Duration</Text>
        <View style={styles.durationInputContainer}>

            style={styles.durationInput}
            value={pastFortnightHours.toString()}
            keyboardType="numeric"
            onChangeText={handlePastFortnightHoursChange}
            maxLength={3}
          />
          <Text style={styles.durationSeparator}>:</Text>
          <TextInput
            style={styles.durationInput}
            value={pastFortnightMinutes.toString()}
            keyboardType="numeric"
            onChangeText={handlePastFortnightMinutesChange}
            maxLength={2}
          />
        </View>
      </View>
      <ScrollView horizontal contentContainerStyle={styles.fortnightsContainer}>
        {fortnights.map((fortnight, index) => (
          <View key={index} style={styles.fortnight}>
